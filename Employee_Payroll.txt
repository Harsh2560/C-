using System;
using System.Collections.Generic;

abstract class Employee
{
    public int EmployeeId { get; set; }
    public string EmployeeName { get; set; }
    private double salary;

    public Employee(int id, string name)
    {
        EmployeeId = id;
        EmployeeName = name;
    }

    protected void SetSalary(double amount)
    {
        salary = amount;
    }

    protected double GetSalary()
    {
        return salary;
    }

    public abstract void CalculateSalary();

    public void DisplayInfo()
    {
        Console.WriteLine("Employee: " + EmployeeName);
        Console.WriteLine("Salary: " + GetSalary());
        Console.WriteLine();
    }
}

class PermanentEmployee : Employee
{
    public PermanentEmployee(int id, string name) : base(id, name)
    {
    }

    public override void CalculateSalary()
    {
        SetSalary(75000);
    }

    public void DisplayType()
    {
        Console.WriteLine("Type: Permanent");
    }
}

class ContractEmployee : Employee
{
    public ContractEmployee(int id, string name) : base(id, name)
    {
    }

    public override void CalculateSalary()
    {
        SetSalary(45000);
    }

    public void DisplayType()
    {
        Console.WriteLine("Type: Contract");
    }
}

class Program
{
    static void Main()
    {
        List<Employee> list = new List<Employee>();

        list.Add(new PermanentEmployee(1, "Krishna"));
        list.Add(new ContractEmployee(2, "Ravi"));

        foreach (Employee e in list)
        {
            e.CalculateSalary();
            e.DisplayInfo();

            if (e is PermanentEmployee)
                ((PermanentEmployee)e).DisplayType();
            else if (e is ContractEmployee)
                ((ContractEmployee)e).DisplayType();

            Console.WriteLine();
        }
    }
}
