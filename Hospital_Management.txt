using System;
using System.Collections.Generic;

abstract class Person
{
    public int Id { get; set; }
    public string Name { get; set; }

    public Person(int id, string name)
    {
        Id = id;
        Name = name;
    }

    public void GetDetails()
    {
        Console.WriteLine("Id: " + Id + ", Name: " + Name);
    }

    public abstract void PerformDuty();
}

class Doctor : Person
{
    public string Department { get; set; }

    public Doctor(int id, string name, string dept) : base(id, name)
    {
        Department = dept;
    }

    public override void PerformDuty()
    {
        Console.WriteLine("Doctor " + Name + " is treating patients.");
    }
}

class Nurse : Person
{
    public string Department { get; set; }

    public Nurse(int id, string name, string dept) : base(id, name)
    {
        Department = dept;
    }

    public override void PerformDuty()
    {
        Console.WriteLine("Nurse " + Name + " is assisting doctors.");
    }
}

class Patient : Person
{
    private string disease;

    public Patient(int id, string name, string d) : base(id, name)
    {
        disease = d;
    }

    public override void PerformDuty()
    {
        Console.WriteLine("Patient " + Name + " is receiving treatment.");
    }
}

class Program
{
    static void Main()
    {
        List<Person> list = new List<Person>();

        list.Add(new Doctor(1, "Krishna", "Cardiology"));
        list.Add(new Nurse(2, "Ravi", "ICU"));
        list.Add(new Patient(3, "Anu", "Fever"));

        foreach (Person p in list)
        {
            p.PerformDuty();
        }
    }
}
